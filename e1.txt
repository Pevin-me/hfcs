clc;
clear;
close all;

fc = 1.5e9;
d0 = 100;
sigma = 3;
d = (1:2:31).^2;

Gt = [1 1 0.5];
Gr = [1 0.5 0.5];
n = [2 3 6];

lambda = 3e8/fc;

for k = 1:3
    free(k,:) = -20*log10(lambda*sqrt(Gt(k)*Gr(k))./(4*pi*d));

    PL0 = -20*log10(lambda/(4*pi*d0));

    logdist(k,:) = PL0 + 10*n(k)*log10(d/d0);

    lognorm(k,:) = PL0 + 10*n(1)*log10(d/d0) ...
                   + sigma*randn(size(d));
end

figure(1);
semilogx(d,free);
grid on;
title('Free Space Path Loss');
xlabel('Distance (m)');
ylabel('Path Loss (dB)');
legend('Case 1','Case 2','Case 3');

figure(2);
semilogx(d,logdist);
grid on;
title('Log Distance Path Loss');
xlabel('Distance (m)');
ylabel('Path Loss (dB)');
legend('n=2','n=3','n=6');

figure(3);
semilogx(d,lognorm);
grid on;
title('Log Normal Shadowing');
xlabel('Distance (m)');
ylabel('Path Loss (dB)');
legend('Case 1','Case 2','Case 3');
