d0 = 100;        % Reference distance in meters

Gt = [1 1 0.5];      % Transmitter gains
Gr = [1 0.5 0.5];    % Receiver gains

%% Loop to Calculate Path Loss for Different Models

    % Free Space Path Loss
    y_Free(k,:) = PL_free(fc, distance, Gt(k), Gr(k));

    % Log-distance Path Loss

    % Log-normal Shadowing Path Loss (constant exponent n = 2)
    y_lognorm(k,:) = PL_logdist_or_norm(fc, distance, d0, Exp(1), sigma);

end

%% ------------------------------------------------------------------------
%% Free Space Path Loss Model
function PL = PL_free(fc, d, Gt, Gr)

    lambda = 3e8 / fc;                  % Wavelength
    tmp = lambda ./ (4 * pi * d);       % Basic path loss equation

    if nargin > 2
        tmp = tmp * sqrt(Gt);           % Incorporate transmitter gain
    end

    if nargin > 3
        tmp = tmp * sqrt(Gr);           % Incorporate receiver gain
    end

    PL = -20 * log10(tmp);              % Convert to dB

end

%% ------------------------------------------------------------------------
%% Log-distance / Log-normal Shadowing Path Loss Model
function PL = PL_logdist_or_norm(fc, d, d0, n, sigma)

    lambda = 3e8 / fc;    % Wavelength

    PL = -20 * log10(lambda / (4 * pi * d0)) + ...
         10 * n * log10(d / d0);

    % Add log-normal shadowing if sigma is provided
    if nargin > 4
        PL = PL + sigma * randn(size(d));
    end

end

%% BER Performance of SC, MRC and EGC over Rayleigh and Ricean Channels

clc;

% Preallocation for Bit Error Rate
Pe_sc  = zeros(12,4,2);   % Selective Combining
Pe_mrc = zeros(12,4,2);   % Maximal Ratio Combining
Pe_egc = zeros(12,4,2);   % Equal Gain Combining

tic

for R = 0:1      % Rayleigh = 0, Ricean = 1

    % Different fading channel has different variance
    if R == 0
        sigma = 1/sqrt(2);
    else
        sigma = 1/2;
    end

    SNR = -10:2:12;

    for enr_index = 1:length(SNR)

        % SNR (Linear)
        enr = 10^(SNR(enr_index)/10);

        % Generate QPSK data (2 bits/symbol)
        data = rand(2, sample_num);
        data = 2*(data > 0.5) - 1;

        Edata = sqrt(2);
        En = Edata / enr;

        for L = 1:4

            % Generate AWGN
            n = normrnd(0, sqrt(En/2), 2, sample_num, L) + ...
                1i*normrnd(0, sqrt(En/2), 2, sample_num, L);

            % Generate fading gain
            g = normrnd(R/2, sigma, 1, sample_num, L) + ...
                1i*normrnd(R/sqrt(2), 1/2, 1, sample_num, L);

            g_tmp = repmat(g, 2, 1, 1);
            tx_data = repmat(data, 1, 1, L);

            % Received signal
            r = g_tmp .* tx_data + n;

            %% Selective Combining
            [Pe_sc(enr_index,L,R+1), result_sc] = ...
                selective_combining(g, g_tmp, r, sample_num, data);

            %% Maximal Ratio Combining
            [Pe_mrc(enr_index,L,R+1), result_mrc] = ...
                maximal_ratio_combining(g_tmp, r, sample_num, data);

            %% Equal Gain Combining
            [Pe_egc(enr_index,L,R+1), result_egc] = ...
                equal_gain_combining(g_tmp, r, sample_num, data);

        end
    end
end

toc

%% =======================================================================
%% Functions
%% =======================================================================

function [BER, result] = selective_combining(g, g_2, r, sample_num, data)

g_abs = abs(g);

[~, I] = max(g_abs, [], 3);

r_sc_tmp = zeros(2, sample_num);
g_sc = zeros(2, sample_num);

for i = 1:sample_num
    r_sc_tmp(:,i) = r(:,i,I(i));
    g_sc(:,i) = g_2(:,i,I(i));
end

r_sc = real(exp(-1i*angle(g_sc)) .* r_sc_tmp);

result = (r_sc > 0)*2 - 1;

BER = get_BER(result, data, sample_num);

end

%% -----------------------------------------------------------------------

function [BER, result] = maximal_ratio_combining(g_2, r, sample_num, data)

r_mrc = real(sum(conj(g_2).*r, 3));

result = (r_mrc > 0)*2 - 1;

BER = get_BER(result, data, sample_num);

end

%% -----------------------------------------------------------------------

function [BER, result] = equal_gain_combining(g_2, r, sample_num, data)

r_egc = real(sum(exp(-1i*angle(g_2)).*r, 3));

result = (r_egc > 0)*2 - 1;

BER = get_BER(result, data, sample_num);

end

%% -----------------------------------------------------------------------

function Pe = get_BER(A, B, sample_num)

wrong = (A ~= B);

error_count = sum(sum(wrong));

Pe = error_count / (sample_num * 2);

end

%% Alamouti OSTBC over Rician Fading Channel
% QPSK and 8-PSK
% 2x1 and 2x2 MIMO
% Compatible with recent MATLAB releases

clear;
close all;
clc;

%% Simulation Parameters
snapshots = 100000;                 % Must be even
if mod(snapshots,2)~=0
    snapshots = snapshots + 1;
end

EbNo = 0:10;
K = [4.0; 0.6];                     % Rician K-factor
M = [4; 8];                         % PSK orders

Mt = 2;                             % Transmit antennas
Mr = [1;2];                         % Receive antennas

%% OSTBC Objects
ostbcEnc = comm.OSTBCEncoder( ...
    'NumTransmitAntennas',Mt);

ric_ber = zeros(length(EbNo),length(M),length(K),length(Mr));
sum_BER = zeros(length(EbNo),length(M),length(K),length(Mr));

%% Main Simulation
for mr = 1:length(Mr)

    ostbcComb = comm.OSTBCCombiner( ...
        'NumTransmitAntennas',Mt,...
        'NumReceiveAntennas',Mr(mr));

    H = zeros(Mr(mr),Mt,snapshots);
    ric_msg = zeros(snapshots,Mr(mr));

    for k = 1:length(K)

        mu = sqrt(K(k)/(K(k)+1));
        s = sqrt(1/(K(k)+1));

        for m = 1:length(M)

            %% Theoretical BER
            ric_ber(:,m,k,mr) = berfading( ...
                EbNo,...
                'psk',...
                M(m),...
                Mr(mr)*Mt,...
                K(k));

            %% Generate Message
            message = randi([0 M(m)-1],snapshots,1);

            %% PSK Modulation
            mod_msg = pskmod(message,M(m),0,'gray');

            Es = 1;

            %% Alamouti Encoding
            alam_msg = step(ostbcEnc,mod_msg);

            %% Rician Channel
            h = mu + s/sqrt(2)*( ...
                randn(Mr(mr),Mt,snapshots/2) + ...
                1i*randn(Mr(mr),Mt,snapshots/2));

            H(:,:,1:2:end-1) = h;
            H(:,:,2:2:end) = h;

            pathGainself = permute(H,[3 2 1]);

            %% Channel Transmission
            for q = 1:snapshots
                ric_msg(q,:) = ( ...
                    sqrt(Es/Mt) * ...
                    H(:,:,q) * ...
                    alam_msg(q,:).' ...
                    ).';
            end

            %% BER Simulation
            BER = zeros(100,length(EbNo));

            snr = EbNo + 10*log10(log2(M(m)));

            for c = 1:100

                for jj = 1:length(EbNo)

                    noisy_mod = awgn( ...
                        ric_msg,...
                        snr(jj),...
                        'measured',...
                        'dB');

                    decodeData = step( ...
                        ostbcComb,...
                        noisy_mod,...
                        pathGainself);

                    demod_msg = pskdemod( ...
                        decodeData,...
                        M(m),...
                        0,...
                        'gray');

                    [~,BER(c,jj)] = biterr(message,demod_msg);

                end

            end

            sum_BER(:,m,k,mr) = mean(BER,1).';

        end
    end
end

clc;
clear;
close all;

%% System Parameters
Nt = 2;                     % Transmit antennas
Nr = 2;                     % Receive antennas
    
EbNoVec = -10:5:20;
modOrd = 2;                 % QPSK (2 bits/symbol)
M = 2^modOrd;

stream = RandStream('mt19937ar');

%% Generate all ML candidates
allBits = de2bi(0:2^(modOrd*Nt)-1, modOrd*Nt, 'left-msb').';
allTxSig = reshape(pskmod(allBits(:), M, 0, 'gray', 'InputType','bit'), ...
                   Nt, []);

%% BER Arrays
BER_ZF = zeros(length(EbNoVec),1);
BER_MMSE = zeros(length(EbNoVec),1);
BER_ML = zeros(length(EbNoVec),1);

%% Simulation Parameters
maxBits = 1e5;
maxErrs = 100;

%% Plot
figure;
hold on;
grid on;
set(gca,'YScale','log');
xlabel('Eb/No (dB)');
ylabel('BER');
title('2x2 Spatial Multiplexing (QPSK)');
ylim([1e-4 1]);

%% Loop over SNR
for idx = 1:length(EbNoVec)

    snrIndB = EbNoVec(idx) + 10*log10(modOrd);
    snrLinear = 10^(snrIndB/10);

    totalBits = 0;

    errZF = 0;
    errMMSE = 0;
    errML = 0;

    while totalBits < maxBits && ...
         (errZF < maxErrs || errMMSE < maxErrs || errML < maxErrs)

        %% Random bits
        msg = randi(stream,[0 1],Nt*modOrd,1);

        %% QPSK Modulation
        txSig = pskmod(msg,M,0,'gray','InputType','bit');

        %% Rayleigh Channel
        H = (randn(stream,Nr,Nt)+1i*randn(stream,Nr,Nt))/sqrt(2);

        %% AWGN
        rxSig = awgn(H*txSig,snrIndB,'measured',stream);

        %% =======================
        %% ZF-SIC
        %% =======================
        r = rxSig;
        Htemp = H;

        estZF = zeros(Nt*modOrd,1);
        orderVec = 1:Nt;

        for n = 1:Nt

            G = (Htemp'*Htemp)\eye(size(Htemp,2));

            [~,k] = min(diag(G));

            symNum = orderVec(k);

            detected = G(k,:)*Htemp'*r;

            bits = pskdemod(detected,M,0,'gray','OutputType','bit');

            estZF((symNum-1)*modOrd+1:symNum*modOrd)=bits;

            if n<Nt
                r = r - Htemp(:,k)*pskmod(bits,M,0,'gray','InputType','bit');
            end

            Htemp(:,k)=[];
            orderVec(k)=[];

        end

        %% =======================
        %% MMSE-SIC
        %% =======================
        r = rxSig;
        Htemp = H;

        estMMSE = zeros(Nt*modOrd,1);
        orderVec = 1:Nt;

        for n = 1:Nt

            G = (Htemp'*Htemp + ...
                (size(Htemp,2)/snrLinear)*eye(size(Htemp,2)))\eye(size(Htemp,2));

            [~,k] = min(diag(G));

            symNum = orderVec(k);

            detected = G(k,:)*Htemp'*r;

            bits = pskdemod(detected,M,0,'gray','OutputType','bit');

            estMMSE((symNum-1)*modOrd+1:symNum*modOrd)=bits;

            if n<Nt
                r = r - Htemp(:,k)*pskmod(bits,M,0,'gray','InputType','bit');
            end

            Htemp(:,k)=[];
            orderVec(k)=[];

        end

        %% =======================
        %% ML Detection
        %% =======================
        metric = sum(abs(rxSig - H*allTxSig).^2,1);

        [~,k] = min(metric);

        estML = allBits(:,k);

        %% BER Update
        errZF = errZF + biterr(msg,estZF);
        errMMSE = errMMSE + biterr(msg,estMMSE);
        errML = errML + biterr(msg,estML);

        totalBits = totalBits + length(msg);

    end

    BER_ZF(idx) = errZF/totalBits;
    BER_MMSE(idx) = errMMSE/totalBits;
    BER_ML(idx) = errML/totalBits;

    fprintf('Eb/No = %2d dB | ZF = %.3e | MMSE = %.3e | ML = %.3e\n',...
        EbNoVec(idx),BER_ZF(idx),BER_MMSE(idx),BER_ML(idx));

end

%% Plot Results
semilogy(EbNoVec,BER_ZF,'r*-','LineWidth',1.5);
semilogy(EbNoVec,BER_MMSE,'bo-','LineWidth',1.5);
semilogy(EbNoVec,BER_ML,'gd-','LineWidth',1.5);

legend('ZF-SIC','MMSE-SIC','ML','Location','southwest');
grid on;

clc;

%% ================================================================
%  802.11n HT 2x2 MIMO Packet Error Rate Simulation
%  20 MHz, MCS 15, TGn Model-B
% ================================================================

%% Create HT configuration
cfgHT = wlanHTConfig;

cfgHT.ChannelBandwidth = 'CBW20';
cfgHT.NumTransmitAntennas = 2;
cfgHT.NumSpaceTimeStreams = 2;
cfgHT.PSDULength = 1000;
cfgHT.MCS = 15;
cfgHT.ChannelCoding = 'BCC';

%% ================================================================
% Configure TGn Channel
% ================================================================

tgnChannel = wlanTGnChannel;

tgnChannel.DelayProfile = 'Model-B';

tgnChannel.NumTransmitAntennas = ...
    cfgHT.NumTransmitAntennas;

tgnChannel.NumReceiveAntennas = 2;

tgnChannel.TransmitReceiveDistance = 10;

tgnChannel.LargeScaleFadingEffect = 'None';

%% ================================================================
% Simulation Parameters
% ================================================================

snr = 10:10:50;

maxNumPEs = 50;
maxNumPackets = 1000;

%% ================================================================
% Sampling Rate
% ================================================================

fs = wlanSampleRate(cfgHT);

%% Set TGn channel sampling rate
tgnChannel.SampleRate = fs;

%% ================================================================
% Get WLAN Field Indices
% ================================================================

ind = wlanFieldIndices(cfgHT);

%% Number of SNR points
S = numel(snr);

%% Preallocate PER array
packetErrorRate = NaN(S,1);

%% ================================================================
% Simulation Loop
% ================================================================

for i = 1:S

    disp(' ');
    disp(['Starting simulation for SNR = ' num2str(snr(i)) ' dB']);

    numPacketErrors = 0;
    n = 1;

    %% ------------------------------------------------------------
    % Loop over packets
    % -------------------------------------------------------------

    while (numPacketErrors <= maxNumPEs) && ...
            (n <= maxNumPackets)

        %% Generate random PSDU

        txPSDU = randi( ...
            [0 1], ...
            cfgHT.PSDULength * 8, ...
            1);

        %% Generate 802.11n HT waveform

        tx = wlanWaveformGenerator( ...
            txPSDU, ...
            cfgHT);

        %% Add trailing zeros
        % Allows for channel filter delay

        tx = [tx; ...
            zeros(15, cfgHT.NumTransmitAntennas)];

        %% --------------------------------------------------------
        % Pass waveform through TGn channel
        % ---------------------------------------------------------

        reset(tgnChannel);

        rxNoNoise = tgnChannel(tx);

        %% --------------------------------------------------------
        % Calculate received signal power
        % ---------------------------------------------------------

        signalPower = mean(abs(rxNoNoise(:)).^2);

        %% --------------------------------------------------------
        % Create AWGN channel
        % ---------------------------------------------------------

        awgnChannel = comm.AWGNChannel( ...
            'NoiseMethod', ...
            'Signal to noise ratio (SNR)', ...
            'SNR', ...
            snr(i), ...
            'SignalPower', ...
            signalPower);

        %% Add AWGN noise

        rx = awgnChannel(rxNoNoise);

        %% ========================================================
        % PACKET DETECTION
        % =========================================================

        coarsePktOffset = wlanPacketDetect( ...
            rx, ...
            cfgHT.ChannelBandwidth);

        %% If packet is not detected

        if isempty(coarsePktOffset)

            numPacketErrors = numPacketErrors + 1;

            n = n + 1;

            continue;

        end

        %% ========================================================
        % COARSE FREQUENCY OFFSET ESTIMATION
        % =========================================================

        lstf = rx( ...
            coarsePktOffset + ...
            (ind.LSTF(1):ind.LSTF(2)), ...
            :);

        coarseFreqOff = wlanCoarseCFOEstimate( ...
            lstf, ...
            cfgHT.ChannelBandwidth);

        %% Correct coarse frequency offset

        rx = frequencyOffset( ...
            rx, ...
            fs, ...
            -coarseFreqOff);

        %% ========================================================
        % FINE PACKET TIMING ESTIMATION
        % =========================================================

        nonhtfields = rx( ...
            coarsePktOffset + ...
            (ind.LSTF(1):ind.LSIG(2)), ...
            :);

        finePktOffset = wlanSymbolTimingEstimate( ...
            nonhtfields, ...
            cfgHT.ChannelBandwidth);

        %% Final packet offset

        pktOffset = coarsePktOffset + finePktOffset;

        %% Check packet offset

        if pktOffset > 15

            numPacketErrors = numPacketErrors + 1;

            n = n + 1;

            continue;

        end

        %% ========================================================
        % FINE FREQUENCY OFFSET ESTIMATION
        % =========================================================

        lltf = rx( ...
            pktOffset + ...
            (ind.LLTF(1):ind.LLTF(2)), ...
            :);

        fineFreqOff = wlanFineCFOEstimate( ...
            lltf, ...
            cfgHT.ChannelBandwidth);

        %% Correct fine frequency offset

        rx = frequencyOffset( ...
            rx, ...
            fs, ...
            -fineFreqOff);

        %% ========================================================
        % HT-LTF EXTRACTION
        % =========================================================

        htltf = rx( ...
            pktOffset + ...
            (ind.HTLTF(1):ind.HTLTF(2)), ...
            :);

        %% ========================================================
        % HT-LTF DEMODULATION
        % =========================================================

        htltfDemod = wlanHTLTFDemodulate( ...
            htltf, ...
            cfgHT);

        %% ========================================================
        % CHANNEL ESTIMATION
        % =========================================================

        chanEst = wlanHTLTFChannelEstimate( ...
            htltfDemod, ...
            cfgHT);

        %% ========================================================
        % EXTRACT HT-DATA
        % =========================================================

        htdata = rx( ...
            pktOffset + ...
            (ind.HTData(1):ind.HTData(2)), ...
            :);

        %% ========================================================
        % NOISE VARIANCE
        % ========================================================
        %
        % SNR = SignalPower / NoisePower
        %
        % Therefore:
        %
        % NoisePower = SignalPower / 10^(SNR/10)
        %
        % This replaces htNoiseEstimate(), which is unavailable
        % in the user's MATLAB installation.
        % =========================================================

        nVarHT = signalPower / ...
            (10^(snr(i)/10));

        %% ========================================================
        % HT DATA RECOVERY
        % =========================================================

        rxPSDU = wlanHTDataRecover( ...
            htdata, ...
            chanEst, ...
            nVarHT, ...
            cfgHT);

        %% ========================================================
        % CHECK PACKET ERROR
        % =========================================================

        packetError = any( ...
            biterr(txPSDU, rxPSDU));

        %% Update packet error count

        numPacketErrors = ...
            numPacketErrors + packetError;

        %% Increment packet counter

        n = n + 1;

    end

    %% ============================================================
    % Calculate Packet Error Rate
    % =============================================================

    packetErrorRate(i) = ...
        numPacketErrors / (n - 1);

    %% Display result

    disp(['SNR = ' num2str(snr(i)) ...
        ' dB | Packets = ' num2str(n-1) ...
        ' | Packet Errors = ' num2str(numPacketErrors) ...
        ' | PER = ' num2str(packetErrorRate(i))]);

end

%% ================================================================
% Plot PER versus SNR
% ================================================================

figure;

semilogy( ...
    snr, ...
    packetErrorRate, ...
    '-ob', ...
    'LineWidth', 1.5);

grid on;

xlabel('SNR [dB]');

ylabel('Packet Error Rate (PER)');

title( ...
    '802.11n 20 MHz, MCS 15, 2x2 TGn Model-B');

xlim([min(snr) max(snr)]);
